procedures_definition(custom_block: resgatar)
    ev3display_setStatusLight(option: "6")
    ev3move_stopMove()
    control_wait(duration: 0.5)
    ev3move_move(value: 0.2, direction: "back", unit: "rotations")
    ev3move_steer(steering: 100, value: 2.1, unit: "rotations")
    ev3move_move(value: 1, direction: "back", unit: "rotations")
    ev3motor_motorSetStopAction(port: "2", option: "1")
    ev3motor_motorTurnForSpeed(port: "2", value: 2, speed: -100, unit: "seconds")
    control_wait(duration: 0.5)
    ev3move_move(value: 1, direction: "forward", unit: "rotations")
    ev3move_steer(steering: -100, value: 2.1, unit: "rotations")
    ev3move_move(value: 0.2, direction: "forward", unit: "rotations")
    data_setvariableto(value: "1", variable: "resgatou")
    ev3display_setStatusLight(option: "4")

        data_setvariableto(value: operator_add(num1: operator_multiply(num1: g_subsensor, num2: 1.8), num2: operator_multiply(num1: operator_subtract(num1: g_subsensor, num2: g_ultimoerro), num2: 0.8)), variable: "correção")

                procedures_call(name: "resgatar")
            else
                procedures_call(name: "seguelinha")

procedures_definition(custom_block: segueLinha)
    control_if_else(condition: operator_and(operand1: operator_gt(operand1: g_sendireito, operand2: "60"), operand2: operator_lt(operand1: g_senesquerdo, operand2: "10")))
        ev3move_steer(steering: "0", value: 0.5, unit: "rotations")
        control_repeat_until(condition: ev3sensors_isEV3ColorSensorReflected(port: "1", value: 10, comparator: "4"))
            ev3move_startDualSpeed(left_speed: -20, right_speed: 20)
        control_repeat_until(condition: ev3sensors_isEV3ColorSensorReflected(port: "1", value: 60, comparator: "2"))
            ev3move_startDualSpeed(left_speed: -20, right_speed: 20)
        ev3move_stopMove()
        data_setvariableto(value: "0", variable: "subSensor")
    else
        control_if_else(condition: operator_and(operand1: operator_lt(operand1: g_sendireito, operand2: "10"), operand2: operator_gt(operand1: g_senesquerdo, operand2: "60")))
            ev3move_steer(steering: "0", value: 0.5, unit: "rotations")
            control_repeat_until(condition: ev3sensors_isEV3ColorSensorReflected(port: "4", value: 10, comparator: "4"))
                ev3move_startDualSpeed(left_speed: 20, right_speed: -20)
            control_repeat_until(condition: ev3sensors_isEV3ColorSensorReflected(port: "4", value: 60, comparator: "2"))
                ev3move_startDualSpeed(left_speed: 20, right_speed: -20)
            ev3move_stopMove()
            data_setvariableto(value: "0", variable: "subSensor")
        else
            ev3move_startDualSpeed(left_speed: operator_add(num1: g_velseguelinha, num2: g_correo), right_speed: operator_subtract(num1: g_velseguelinha, num2: g_correo))

procedures_definition(custom_block: desviar)
    ev3display_setStatusLight(option: "5")
    ev3move_stopMove()
    ev3move_move(value: 0.2, direction: "back", unit: "rotations")
    ev3move_steer(steering: "direita: 100", value: 0.95, unit: "rotations")
    ev3move_steer(steering: 0, value: 1.3, unit: "rotations")
    ev3move_steer(steering: "esquerda: -100", value: 0.95, unit: "rotations")
    ev3move_steer(steering: "0", value: 2, unit: "rotations")
    ev3move_steer(steering: "-100", value: 0.95, unit: "rotations")
    control_repeat_until(condition: operator_or(operand1: ev3sensors_isEV3ColorSensorReflected(port: "1", value: 20, comparator: "4"), operand2: ev3sensors_isEV3ColorSensorReflected(port: "4", value: 20, comparator: "4")))
        ev3move_startSteerAtSpeed(steering: "0", speed: 20)
    ev3move_stopMove()
    ev3move_steerDistanceAtSpeed(steering: "0", value: 0.4, speed: 20, unit: "rotations")
    control_repeat_until(condition: ev3sensors_isEV3ColorSensorReflected(port: "1", value: 20, comparator: "4"))
        ev3move_startSteerAtSpeed(steering: "direita: 100", speed: 20)
    ev3move_steerDistanceAtSpeed(steering: "esquerda: -100", value: 0.2, speed: 20, unit: "rotations")
    ev3move_stopMove()
    ev3display_setStatusLight(option: "4")
